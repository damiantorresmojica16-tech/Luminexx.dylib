import os
import shutil
from fastapi import FastAPI, HTTPException, UploadFile, File, Form, Query
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import FileResponse, JSONResponse

app = FastAPI(title="Widman iOS API - TXT Mode")

# Habilitar CORS para permitir peticiones desde Netlify
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

KEYS_FILE = "keys.txt"

def verificar_key_en_txt(key_a_buscar: str) -> bool:
    """Lee el archivo keys.txt línea por línea y busca la key."""
    if not os.path.exists(KEYS_FILE):
        return False
    
    with open(KEYS_FILE, "r", encoding="utf-8") as f:
        for linea in f:
            # Limpiamos espacios y saltos de línea
            if linea.strip() == key_a_buscar.strip():
                return True
    return False

@app.get("/")
def inicio():
    return {"status": "ok", "mensaje": "API de Widman iOS activa (Modo TXT)"}

@app.get("/validar-key")
def validar_key(key: str = Query(...)):
    try:
        key_limpia = key.strip()
        if not key_limpia:
            return JSONResponse(status_code=200, content={"valida": False, "error": "Debes ingresar una Key."})

        # Verificar si la key existe en el archivo keys.txt
        existe = verificar_key_en_txt(key_limpia)

        if not existe:
            return JSONResponse(status_code=200, content={"valida": False, "error": "La Key ingresada no existe en el archivo."})

        return JSONResponse(status_code=200, content={"valida": True, "mensaje": "Key correcta"})
    except Exception as e:
        return JSONResponse(status_code=500, content={"valida": False, "error": f"Error en servidor: {str(e)}"})

@app.post("/modificar-archivo")
async def modificar_archivo(key: str = Form(...), file: UploadFile = File(...)):
    # Validar nuevamente antes de procesar el archivo
    if not verificar_key_en_txt(key.strip()):
        raise HTTPException(status_code=403, detail="Key no autorizada o inexistente.")

    input_filename = f"temp_{file.filename}"
    output_filename = f"modificado_{file.filename}"

    try:
        with open(input_filename, "wb") as buffer:
            shutil.copyfileobj(file.file, buffer)

        # Proceso de modificación/copia
        shutil.copy(input_filename, output_filename)

        return FileResponse(
            path=output_filename,
            filename=f"modificado_{file.filename}",
            media_type='application/octet-stream'
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Error al procesar: {str(e)}")
    finally:
        if os.path.exists(input_filename):
            os.remove(input_filename)
